package com.yourcompany.payroll.model;

import jakarta.persistence.*;

@Entity
@Table(name = "payroll_batches") // Use snake_case for table names typically
public class Batch {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(nullable = false, length = 100)
    private String name;

    @Column(name = "payment_count")
    private int paymentCount = 0; // Default to 0

    @Column(name = "last_payment_date")
    private LocalDateTime lastPaymentDate;

    @Column(name = "payment_status", length = 20)
    private String paymentStatus = "Draft"; // Default status

    // --- Relationship ---
    // MappedBy indicates the field in the Employee entity that owns the relationship
    // CascadeType.PERSIST and MERGE are often useful, REMOVE can be dangerous if employees should exist without a batch
    // FetchType.LAZY is generally recommended for performance
    @OneToMany(mappedBy = "batch", cascade = {CascadeType.PERSIST, CascadeType.MERGE}, fetch = FetchType.LAZY)
    private List<Employee> employees = new ArrayList<>();

    // --- Constructors ---
    public Batch() {
    }

    public Batch(String name) {
        this.name = name;
    }

    // --- Getters and Setters (NO Lombok) ---

    public Long getId() {
        return id;

        this.id = id;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public int getPaymentCount() {
        return paymentCount;
    }

    public void setPaymentCount(int paymentCount) {
        this.paymentCount = paymentCount;
    }

    public LocalDateTime getLastPaymentDate() {
        return lastPaymentDate;
    }

    public void setLastPaymentDate(LocalDateTime lastPaymentDate) {
        this.lastPaymentDate = lastPaymentDate;
    }

    public String getPaymentStatus() {
        return paymentStatus;
    }

    public void setPaymentStatus(String paymentStatus) {
        this.paymentStatus = paymentStatus;
    }

    public List<Employee> getEmployees() {
        return employees;
    }

    public void setEmployees(List<Employee> employees) {
        this.employees = employees;
    }

    // --- Convenience Methods (Optional but helpful) ---
    public void addEmployee(Employee employee) {
        this.employees.add(employee);
        employee.setBatch(this);
    }

    public void removeEmployee(Employee employee) {
        this.employees.remove(employee);
        employee.setBatch(null);
    }

    // --- toString, equals, hashCode (Optional but good practice) ---
    @Override
    public String toString() {
        return "Batch{" +
               "id=" + id +
               ", name='" + name + '\'' +
               ", paymentCount=" + paymentCount +
               ", lastPaymentDate=" + lastPaymentDate +
               ", paymentStatus='" + paymentStatus + '\'' +
               '}';
    }

    // Basic equals/hashCode based on ID
    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        Batch batch = (Batch) o;
        return id != null && id.equals(batch.id);
    }

    @Override
    public int hashCode() {
        return getClass().hashCode(); // Or use Objects.hash(id) if ID is set
    }

package com.yourcompany.payroll.model;

@Entity
@Table(name = "employees")

    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(nullable = false, length = 100)
    private String name;

    // Use BigDecimal for currency to avoid floating-point issues
    @Column(name = "salary_amount", nullable = false, precision = 10, scale = 2)
    private BigDecimal salaryAmount;

    @Column(name = "bank_details", nullable = false, length = 100)
    private String bankDetails;

    @Column(name = "payment_ref", length = 50)
    private String paymentRef;

    @Column(name = "your_ref", length = 50)
    private String yourRef;

    @Column(length = 255) // Text notes can be longer
    private String notes;

    // --- Relationship ---
    // Many employees can belong to one batch
    // FetchType.LAZY is preferred here
    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "batch_id") // Name of the foreign key column in the employees table
    private Batch batch; // This field owns the relationship

    // --- Constructors ---
    public Employee() {
    }

        this.name = name;
        this.salaryAmount = salaryAmount;

    // --- Payment Initiation Endpoint (MODIFIED) ---

    // @Operation(summary = "Initiate payment for a batch, setting status to Pending")
    @PostMapping("/batches/{id}/initiate-payment")
    @Transactional
    public ResponseEntity<BatchDto> initiatePayment(@PathVariable Long id, @RequestBody PaymentDetailsDto paymentDetails) {
        // Validation for payload
         if (paymentDetails.getDebitAccount() == null || paymentDetails.getDebitAccount().trim().isEmpty()) {
            return ResponseEntity.badRequest().body(null); // Or return an error DTO
        }
        // Optional: Validate that the selected debit account exists in the DB
        bankAccountRepository.findByAccountNumber(paymentDetails.getDebitAccount())
             .orElseThrow(() -> new EntityNotFoundException("Selected Debit Account not found: " + paymentDetails.getDebitAccount()));


        Batch batch = batchRepository.findById(id)
                .orElseThrow(() -> new EntityNotFoundException("Batch not found with id: " + id));

        // Check if batch has employees
        long employeeCount = employeeRepository.countByBatchId(id);
        if (employeeCount == 0) {
            System.err.println("Attempted to initiate payment for empty batch: " + id);
            // Return a specific error response DTO if needed
             return ResponseEntity.status(HttpStatus.CONFLICT).build(); // Conflict - cannot pay empty batch
        }

        // --- MODIFIED STATUS LOGIC ---
        // Update batch status to Pending, increment count, set date
        batch.setPaymentCount(batch.getPaymentCount() + 1);
        batch.setLastPaymentDate(LocalDateTime.now());
        batch.setPaymentStatus("Pending"); // Set status to Pending

        // Here you might trigger notifications or other workflows related to the pending status
        System.out.println("Payment initiated (Status: Pending) for batch " + id + " from account " + paymentDetails.getDebitAccount() +
                           " type: " + paymentDetails.getPayrollType() + " currency: " + paymentDetails.getCurrency());
        // NO actual payment processing simulation here anymore

        Batch updatedBatch = batchRepository.save(batch);
        return ResponseEntity.ok(convertToBatchDto(updatedBatch));
    }

    // --- Bank Accounts Endpoint (MODIFIED) ---

    // @Operation(summary = "Get all bank accounts from database")
    @GetMapping("/bank-accounts")
    public ResponseEntity<List<BankAccountDto>> getBankAccounts() {
        List<BankAccount> accounts = bankAccountRepository.findAll();
        List<BankAccountDto> accountDtos = accounts.stream()
                .map(this::convertToBankAccountDto) // Use helper method
                .collect(Collectors.toList());
        return ResponseEntity.ok(accountDtos);
    }


    // --- Helper Methods for DTO Conversion ---

    private BatchDto convertToBatchDto(Batch batch) {
        int employeeCount = employeeRepository.countByBatchId(batch.getId());
        return new BatchDto(
                batch.getId(),
                batch.getName(),
                batch.getPaymentCount(),
                batch.getLastPaymentDate(),
                batch.getPaymentStatus(),
                employeeCount
        );
    }

    private EmployeeDto convertToEmployeeDto(Employee employee) {
        Long batchId = (employee.getBatch() != null) ? employee.getBatch().getId() : null;
        return new EmployeeDto(
                employee.getId(),
                employee.getName(),
                employee.getSalaryAmount(),
                employee.getBankDetails(),
                employee.getPaymentRef(),
                employee.getYourRef(),
                employee.getNotes(),
                batchId
        );
    }

    // --- NEW Helper Method for BankAccount DTO ---
    private BankAccountDto convertToBankAccountDto(BankAccount account) {
        return new BankAccountDto(
            account.getId(),
            account.getAccountNumber(),
            account.getAccountName(),
            account.getBalance()
        );
    }

     // --- Exception Handling (Basic Example) ---
